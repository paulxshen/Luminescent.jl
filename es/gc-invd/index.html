Acoplador de red diseñado inversamente: región de diseño cuadrada

Diseño inverso de un acoplador de red sobre SOI para una fibra inclinada en una región de diseño cuadrada de 20 x 20 um alimentada directamente por la guía de onda, partiendo de un estado completamente relleno. (el diseño inverso no está disponible en la versión gratuita)

import os
import numpy as np
import matplotlib.pyplot as plt

import gdsfactory as gf
from gdsfactory.technology import LogicalLayer, LayerLevel, LayerStack
import luminescent as lumi
lumi.app()
# a()
here = os.path.dirname(os.path.abspath(__file__))

path = os.path.join(here, "runs", "GC_invd")

wl = 1.55
bandwidth = 0.01
nwls = 3
wls = np.linspace(wl - bandwidth / 2, wl + bandwidth / 2, nwls)

# simulation parameters
nres = 8
boundaries = ["PML", ["PEC", "PML"], "PML"]
PML_decay = [0.001, 0.01, 0.001]
courant = 0.9
subpixel_smoothing = 1
shutoff = None

# horizontal dimensions
D = 10.4  # mode field diameter, SMF-28
R = 1.5 * D / 2  # beam frame half width

S = 20.0  # square design region side, centered at x=0 under beam
x_offset = 0.0  # fiber shift along x from design center
w_wg = 0.5
l_wg = 2.0

nox = lumi.MATERIALS["SiO2"]["n"]
# vertical dimensions
theta = 7.978 * np.pi / 180  # beam angle
h_wg = 0.22
h_etch = 0.08
h_slab = h_wg - h_etch
h_box = 2.0  # buried oxide
h_sub = 0.3  # substrate thickness in domain, continues into PML

margin =zmargin = 1.25
source_offset = 0.25
z_port = h_wg + zmargin

zmin = -h_box - h_sub
zmax = z_port + source_offset

# movie parameters
shot_every = 10

views = [
    lumi.view("Ey", y=0, mirrors=["y-"]),
    lumi.view("Ey", z=h_slab + h_etch / 2, mirrors=["y-"]),
]

# materials
core = "Si"
clad = "SiO2"
n_core = lumi.MATERIALS[core]["n"]
n_clad = lumi.MATERIALS[clad]["n"]
lambda_core = wl / n_core
dx = lambda_core / nres

background = clad
materials = {**lumi.MATERIALS, "design": lumi.placeholder_material(n_core)}

# layer stack: lower mesh order layers override higher ones
WG = (1, 0)
CLAD = (2, 0)
SLAB = (102, 0)
DESIGN = (1001, 0)
CONTOUR = (1002, 0)

layers = {
    "design": LayerLevel(
        layer=LogicalLayer(layer=DESIGN),
        zmin=h_slab,
        thickness=h_etch,
        material="design",
        mesh_order=1,
    ),
    "slab": LayerLevel(
        layer=LogicalLayer(layer=SLAB),
        zmin=0,
        thickness=h_slab,
        material=core,
        mesh_order=102,
    ),
    "core": LayerLevel(
        layer=LogicalLayer(layer=WG),
        zmin=0,
        thickness=h_wg,
        material=core,
        mesh_order=200,
    ),
    "substrate": LayerLevel(
        layer=LogicalLayer(layer=CLAD),
        zmin=zmin,
        thickness=h_sub,
        material=core,
        mesh_order=300,
    ),
}
layer_stack = LayerStack(layers=layers)
# component to optimize: upper half, mirrored about y=0
x_design = -S / 2
p = [(x_design, 0), (x_design, S / 2), (S / 2, S / 2), (S / 2, 0)]  # upper half of square

c = gf.Component()
c.add_polygon(p, layer=DESIGN)
c.add_polygon(p, layer=WG)  # initial design: filled
c.add_polygon(p, layer=CONTOUR)  # design mask
c.add_polygon(p, layer=SLAB)

x_wg = x_design - l_wg
c.add_polygon([(x_wg, 0), (x_wg, w_wg / 2), (x_design, w_wg / 2), (x_design, 0)], layer=WG)

x_fiber = x_offset + np.sin(theta) * (zmargin + source_offset)
c << gf.components.bbox(  # enclose beam frame
    component=c,
    layer=CLAD,
    top=margin,
    right=max(0, x_fiber + R - c.xmax),
)

c.add_port("o1", center=(x_wg, 0), width=w_wg / 2, orientation=180, layer=WG)

c.plot()
frames = [
    lumi.plane_frame(
        "wg",
        start=[-w_wg / 2 - margin, -zmargin],
        stop=[0, h_wg + zmargin],
        boundaries=[("PML", "PEC"), "PML"],
        wls_mode_solver=[wl],
    ),
    lumi.plane_frame(
        "beam",
        modes=[
            [
                lumi.gaussian_mode(
                    D,
                    n_clad,
                    dx=dx,
                    theta=theta,
                    wl=wl,
                    half="y+",
                    Dlim=2 * R,
                )
            ]
        ],
        start=[-R, 0],
        stop=[R, R],
    ),
]

origin = [x_fiber, 0, zmax]  # source above beam monitor, on beam axis
basis = [[1, 0, 0], [0, 1, 0], [0, 0, 1]]
sources = [
    lumi.source(
        "o2",
        origin,
        basis,
        wl=wl,
        bandwidth=wl / 8,
        frame="beam",
    )
]

origin = [x_offset + np.sin(theta) * zmargin, 0, z_port]  # beam monitor below source
monitors = [
    lumi.gf_monitor("o1", c, layer_stack, frame="wg"),
    lumi.monitor("o2", origin, basis, frame="beam"),
]

designs = [
    lumi.design(
        name="design",
        lmin=0.15,
        layer=DESIGN,
        contour_layer=CONTOUR,
        fill_material=core,
        void_material=clad,
        component=c,
        layer_stack=layer_stack,
        materials=materials,
        repdims=[3],
        anchordims=[-2],
    )
]

optimizer = lumi.optimizer(
    # stoploss=0.6,
    # iters=20,
    contrasts=[.4,.7,1],
    rates=[.04,.04,.02],
    momentum=0.7,
    init_dA=0.005,
)
targets = [
    lumi.target(src="o2@0-", dst="o1@0+", target=1, weight=1 / nwls, wl=wl)
    for wl in wls
]

tmax = 1.1*(
    1 / wl * (
        (zmargin + source_offset + 2 * h_box) / np.cos(theta) * n_clad
        + (S + S / 2 + l_wg) * n_core
    )
    + sources[0]["duration"]
)

bbox = c.bbox_np().tolist()
bbox[0].append(zmin)
bbox[1].append(zmax)
c.flatten()
lumi.mesh(path, c, layer_stack, materials, zmin, zmax)
lumi.setup(
    path=path,
    wl=wl,
    wls=wls,
    frames=frames,
    monitors=monitors,
    sources=sources,
    boundaries=boundaries,
    bbox=bbox,
    materials=materials,
    background=background,
    nres=nres,
    tmax=tmax,
    PML_decay=PML_decay,
    courant=courant,
    subpixel_smoothing=subpixel_smoothing,
    shutoff=shutoff,
    shot_every=shot_every,
    views=views,
    designs=designs,
    targets=targets,
    optimizer=optimizer,
)
lumi.solve(path)
import json
hist = json.load(open(os.path.join(path, "history.json")))["history"]
print(f"loss: {hist[0]['loss']:.3f} (iter 1) -> {hist[-1]['loss']:.3f} (final, iter {hist[-1]['iter']})")
loss: 1.000 (iter 1) -> 0.469 (final, iter 67)
plt.plot([h["iter"] for h in hist], [h["loss"] for h in hist])
plt.xlabel("iteration")
plt.ylabel("loss")
plt.show()
mask = np.load(os.path.join(path, "designs", "design", "mask.npy"))  # optimized design: True = core
plt.imshow(mask.T, origin="lower", cmap="gray")
plt.show()
