Curva de guía de onda: inicio rápido

Parámetros S dependientes de la longitud de onda para una curva de guía de onda de Euler en SOI

import os
import numpy as np
import matplotlib.pyplot as plt

import gdsfactory as gf
from gdsfactory.technology import LogicalLayer, LayerLevel, LayerStack
import luminescent as lumi
lumi.app()
here = os.path.dirname(os.path.abspath(__file__))

# simulation folder
path = os.path.join(here, "runs", "bend")
lumi.app(path)
# a()
# raise ValueError(f"Delete {path} before running again")

# dims [um]
# length units are arbitrary so long as they're consistent
r=5
w_wg = 0.5
h_wg = 0.22
bend_type = "euler"

# wavelengths [um]. 
# spectrum computed from single FDTD run. more wavelengths slightly increases runtime
wavelength = 1.55
bandwidth = 0.1
nwls = 11
wavelengths = np.linspace(wavelength - bandwidth / 2, wavelength + bandwidth / 2, nwls)

# "euler" or "circular"
def setup(path,wavelength,wavelengths, r,w_wg,h_wg,bend_type):
    # simulation parameters
    nres = 6
    subpixel_smoothing = 1
    shutoff = 1e-3
    tmax = 100
    PML_decay = [0.001, 0.001, 0.1]  # x y z
    views = [lumi.view("Hz", y="half"), lumi.view("Hz", z=h_wg / 2)]
    shot_every = 10 # movie [periods]

    boundaries = ["PML", "PML", "PML"]

    # margins
    zmargin =margin = 1
    source_offset = 1

    # vertical dims
    zmin = -zmargin
    zmax = h_wg + zmargin

    # materials
    core = "Si"
    clad = "SiO2"
    background = clad

    # layers
    WG = 1, 0
    CLAD = 2, 0

    layer_stack =     LayerStack(
            layers={
                "core": LayerLevel(
                    layer=LogicalLayer(layer=WG),
                    thickness=h_wg,
                    zmin=0,
                    material=core,
                    mesh_order=1,
                )
            }
        )

    # make component in gdsfactory. alternatively can import .gds into gdsfactory
    c = gf.Component()

    # Euler or circular bend
    if bend_type == "euler":
        b = c << gf.components.bend_euler(r)
    elif bend_type == "circular":
        b = c << gf.components.bend_circular(r)
    s = c << gf.components.straight(length=source_offset, width=w_wg)

    s.connect("o2", b.ports["o1"])
    c << gf.components.bbox(component=c, layer=CLAD, right=margin, bottom=margin)

    c.add_port("o1", port=b.ports["o1"])
    c.add_port("o2", port=b.ports["o2"])
    # c.plot()

    bbox = c.bbox_np().tolist()
    bbox[0].append(zmin)
    bbox[1].append(zmax)

    # sources, monitors
    w_port = w_wg + 2 * margin
    frames = [
        lumi.plane_frame(
            "wg_mode",
            start=[-w_port / 2, zmin],
            stop=[w_port / 2, zmax],
            boundaries=["PML", "PML"],
            nmodes=1,
            modes="auto",
            wls_mode_solver=[wavelength],
        ),
        lumi.points_frame("wg_signal", points=[[0, 0]], time_domain=True),
    ]

    sources = [
        lumi.gf_source(
            "o1", c, layer_stack, source_offset,
            frame="wg_mode",
            wl=wavelength,
            bandwidth=max(max(wavelengths) - min(wavelengths), wavelength / 8),
        )
    ]

    monitors = [
        # modal monitors
        lumi.gf_monitor(
            "o1", c, layer_stack,
            frame="wg_mode",
        ),
        lumi.gf_monitor(
            "o2", c, layer_stack,
            frame="wg_mode",
            mirror="x",
        ),
        # time traces (not necessary for S-parameters but for visualizing the signal)
        # lumi.gf_monitor("o2", c, layer_stack, name="o2_signal", frame="wg_signal"),
    ]

    # writes .obj mesh files from gdsfactory component
    c.flatten()
    c.plot(return_fig=True).savefig(os.path.join(here, "layout.png"))
    
    lumi.mesh(path, c, layer_stack, lumi.MATERIALS, zmin, zmax)

    # writes setup file
    lumi.setup(
        path=path,
        wl=wavelength,
        wls=wavelengths,
        materials=lumi.MATERIALS,
        background=background,
        boundaries=boundaries,
        bbox=bbox,
        #
        frames=frames,
        sources=sources,
        monitors=monitors,
        #
        nres=nres,
        subpixel_smoothing=subpixel_smoothing,
        tmax=tmax,
        PML_decay=PML_decay,
        #
        views=views,
        shot_every=shot_every,
        shutoff=shutoff,
    )
    # return c

setup(path, wavelength, wavelengths, r, w_wg, h_wg,bend_type)
lumi.solve(path)
sol = lumi.load(path)
m = sol["monitors"]
mid = nwls // 2
S21 = m["o2"]["modal"][0]["+"][mid] / m["o1"]["modal"][0]["-"][mid]
S11 = m["o1"]["modal"][0]["+"][mid] / m["o1"]["modal"][0]["-"][mid]
print(f"at {wavelength} um: |S21|^2 = {abs(S21)**2:.4f}, |S11|^2 = {abs(S11)**2:.2e}")
at 1.55 um: |S21|^2 = 0.9981, |S11|^2 = 4.10e-04
